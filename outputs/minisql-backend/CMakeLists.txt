cmake_minimum_required(VERSION 3.25)

# Feature selection must happen before the vcpkg toolchain is loaded by project().
option(BUILD_TESTING "Build MiniSQL tests" ON)
if(BUILD_TESTING)
    list(APPEND VCPKG_MANIFEST_FEATURES tests)
endif()
project(MiniSQL VERSION 0.1.0 LANGUAGES CXX)

# Must be set before any target uses C++17/20-only headers, then re-applied
# to targets created earlier in this file.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(BUILD_TESTING)
    add_executable(minisql_arithmetic64_probe tests/arithmetic64_probe.cpp)
    target_link_libraries(minisql_arithmetic64_probe PRIVATE minisql_common)
endif()

find_package(fmt CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(CLI11 CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED COMPONENTS multiprecision)

add_library(minisql_options INTERFACE)
if(MSVC)
    target_compile_options(minisql_options INTERFACE /W4 /permissive- /utf-8)
    target_compile_definitions(minisql_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
    # 第十七章 REQ-CORE-001：表达式嵌套上限为 256，而 MSVC 默认主线程栈只有 1 MiB。
    # 解析/语义/优化链路每层嵌套约需数 KiB，默认栈会在保护阈值之前就溢出，
    # 使深嵌套输入变成进程崩溃（而不是 SyntaxError 2002）。预留 16 MiB 给足余量。
    target_link_options(minisql_options INTERFACE /STACK:16777216)
else()
    target_compile_options(minisql_options INTERFACE -Wall -Wextra -Wpedantic)
endif()

configure_file(cmake/version.hpp.in generated/minisql/common/version.hpp @ONLY)
add_library(minisql_common STATIC
    src/common/error.cpp
    src/common/config.cpp
    src/common/logger.cpp
    src/security/access_catalog.cpp)
target_include_directories(minisql_common PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(minisql_common PUBLIC minisql_options fmt::fmt
    spdlog::spdlog nlohmann_json::nlohmann_json Boost::multiprecision)

# Reserved module targets: these contain no database implementation yet.
foreach(module IN ITEMS transaction)
    add_library(minisql_${module} INTERFACE)
    target_link_libraries(minisql_${module} INTERFACE minisql_common)
endforeach()

add_library(minisql_storage STATIC src/storage/page.cpp src/storage/file_io.cpp src/storage/page_file.cpp src/storage/buffer_pool.cpp src/storage/heap.cpp src/storage/bplus_tree.cpp src/storage/page_bplus_tree.cpp)
target_link_libraries(minisql_storage PUBLIC minisql_common)

add_library(minisql_catalog STATIC src/catalog/catalog.cpp)
target_link_libraries(minisql_catalog PUBLIC minisql_sql)
add_library(minisql_persistent_catalog STATIC src/catalog/persistent_catalog.cpp)
target_link_libraries(minisql_persistent_catalog PUBLIC minisql_catalog minisql_storage)

add_library(minisql_sql STATIC src/sql/lexer.cpp src/sql/parser.cpp)
target_link_libraries(minisql_sql PUBLIC minisql_common)
add_executable(minisql_lexer src/server/lexer_main.cpp)
target_link_libraries(minisql_lexer PRIVATE minisql_sql)
add_executable(minisql_compile src/server/compile_main.cpp)
# X25 绑定层：名称解析的唯一归属地。planner/optimizer/security 都消费它的输出。
add_library(minisql_binder STATIC src/sql/binder.cpp)
target_link_libraries(minisql_binder PUBLIC minisql_catalog)
add_library(minisql_planner STATIC src/sql/planner.cpp)
target_link_libraries(minisql_planner PUBLIC minisql_catalog minisql_binder)
add_library(minisql_optimizer STATIC src/optimizer/optimizer.cpp)
target_link_libraries(minisql_optimizer PUBLIC minisql_planner)
target_link_libraries(minisql_compile PRIVATE minisql_optimizer)
if(WIN32)
    target_link_libraries(minisql_compile PRIVATE shell32)
endif()
add_library(minisql_execution STATIC src/execution/database.cpp src/execution/executor.cpp)
target_link_libraries(minisql_execution PUBLIC minisql_optimizer minisql_persistent_catalog)
add_executable(minisql_database src/server/database_main.cpp)
target_link_libraries(minisql_database PRIVATE minisql_execution)
if(WIN32)
    target_link_libraries(minisql_database PRIVATE shell32)
endif()

add_library(minisql_server STATIC src/server/command_line.cpp src/server/application.cpp)
target_link_libraries(minisql_server PUBLIC minisql_common minisql_execution PRIVATE CLI11::CLI11)
add_executable(minisql src/main.cpp)
target_link_libraries(minisql PRIVATE minisql_server)
