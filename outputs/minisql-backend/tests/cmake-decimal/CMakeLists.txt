cmake_minimum_required(VERSION 3.25)
project(MiniSQLDecimalContract LANGUAGES CXX)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED COMPONENTS multiprecision)
add_executable(decimal_contract ../decimal_contract.cpp ../../src/common/error.cpp)
target_include_directories(decimal_contract PRIVATE ../../include)
target_compile_features(decimal_contract PRIVATE cxx_std_20)
target_link_libraries(decimal_contract PRIVATE nlohmann_json::nlohmann_json Boost::multiprecision)
if(MSVC)
    target_compile_options(decimal_contract PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(decimal_contract PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    target_compile_options(decimal_contract PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME exact_decimal COMMAND decimal_contract)
